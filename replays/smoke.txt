# Smoke replay: the two reference views plus a move and a time change, with the
# cheap invariants that break first when terrain or streaming changes.
#
#   tools/render.sh replay replays/smoke.txt      renders the shots below
#   tools/test.sh                                 same script, rendering off
#
# Seed and size come from the command line (--seed/--size): the world is built
# before the script runs, so `seed` is not a replay command.

# default view, aimed automatically at the peak.
# mesh counters only exist once the streamer has built something, so the
# shot comes first: it blocks until the world is ready.
wait 0.5
assert trees > 1000
# structural health, not looks: no chunk may drop geometry for lack of vertex
# budget, and no generated field may contain a NaN. Both render silently.
assert nonfinite == 0
assert overflow == 0          # no chunk dropped vertices for lack of budget
assert trioverflow == 0       # no chunk lost triangles to the Delaunay array
assert ptoverflow == 0        # no chunk hit the point budget
shot build/shots/smoke-default.png
assert tris > 1000
# Nothing a driver or sokol says at warning level may pass unnoticed: the two
# real defects of the GL port were both just log lines, and pool exhaustion is a
# *warning* (it hands back an invalid handle and aborts later inside a draw).
assert warned == 0
# The vertex budget is a hard cap that silently truncates a chunk's geometry.
assert chunkverts < 100
# Every world builds inside its budget; the number is an alarm, not a target.
assert buildms < 2000
assert chunks_drawn > 20
assert shadowed_pct < 100
assert overflow == 0          # the streamer built 200+ chunks and dropped nothing

# a fixed view from higher up, late afternoon
cam 0,420,900
look 0 -18
time 17.5
shot build/shots/smoke-high.png
assert cam_z > 800
assert chunks_drawn > 20

# walk forward for a second at 60Hz: the camera must move, and stay above the
# ground (the clamp in sim_step is the only thing that keeps it there)
cam 0,180,600
look 0 -6
key W hold 1.0
assert cam_z > 605          # W walks forward: +z at yaw 0
assert cam_y > 0            # never below the ground clamp

# scripted move is smoothed over its duration, not teleported
move 0,180,200 over 1.0
wait 0.5
assert cam_z < 500
shot build/shots/smoke-move.png
